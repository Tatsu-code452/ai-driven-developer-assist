import path from "path";

import {
  parseMarkdownContent
} from "@/services/markdownParser/parseMarkdownContent";

import {
  extractTasks
} from "@/services/TaskExtractor";

import {
  extractDocuments
} from "@/services/DocumentExtractor";

import {
  exportTasks,
  exportDocuments
} from "@/services/JsonExporter";

export const convertMarkdown =
(
  markdown: string,
  outputRoot = "data"
) => {

  const document =
    parseMarkdownContent(
      markdown
    );

  const tasks =
    extractTasks(
      document
    );

  const documents =
    extractDocuments(
      document
    );

  const outputDir =
    path.join(
      outputRoot,
      "generated"
    );

  const tasksFile =
    exportTasks(
      tasks,
      outputDir
    );

  const documentsFile =
    exportDocuments(
      documents,
      outputDir
    );

  return {

    document,

    tasks,

    documents,

    output: {

      tasksFile,

      documentsFile

    },

    summary: {

      taskCount:
        tasks.length,

      documentCount:
        documents.length

    }

  };

};

import type { Document } from "../models/Document";
import type { DocumentModel } from "../models/DocumentModel";

export const extractDocuments = (
  document: DocumentModel
): Document[] => {

  const documentType =
    detectDocumentType(
      document
    );

  return [
    {
      id:
        createDocumentId(
          documentType,
          1
        ),
      title:
        document.title,
      type:
        documentType,
      path:
        document.fileName,
      source:
        document.title
    }
  ];
};

export const detectDocumentType =
(
  document: DocumentModel
) => {

  const title =
    document.title
      .toLowerCase();

  const fileName =
    document.fileName
      .toLowerCase();

  const value =
    title + " " +
    fileName;

  if (
    value.includes("要件") ||
    value.includes(
      "requirements"
    )
  ) {
    return "REQ";
  }

  if (
    value.includes(
      "基本設計"
    ) ||
    value.includes(
      "アーキテクチャ"
    ) ||
    value.includes(
      "design"
    )
  ) {
    return "DES";
  }

  if (
    value.includes(
      "詳細設計"
    )
  ) {
    return "DET";
  }

  if (
    value.includes(
      "テスト"
    )
  ) {
    return "TST";
  }

  if (
    value.includes(
      "運用"
    )
  ) {
    return "OPS";
  }

  return "DOC";
};

export const createDocumentId =
(
  prefix: string,
  sequence: number
): string => {

  return (
    prefix +
    "-" +
    String(
      sequence
    ).padStart(
      3,
      "0"
    )
  );
};

import fs from "fs";
import path from "path";

type ExportOptions<T> = {
  outputDir: string;
  fileName: string;
  data: T;
};

const ensureDirectory = (
  directory: string
): void => {

  if (
    fs.existsSync(directory)
  ) {
    return;
  }

  fs.mkdirSync(
    directory,
    {
      recursive: true
    }
  );
};

export const exportJson = <
  T
>(
  options:
    ExportOptions<T>
): string => {

  ensureDirectory(
    options.outputDir
  );

  const filePath =
    path.join(
      options.outputDir,
      options.fileName
    );

  fs.writeFileSync(
    filePath,
    JSON.stringify(
      options.data,
      null,
      2
    ),
    "utf-8"
  );

  return filePath;
};

import { MarkdownList } from "@/models/MarkdownList";
import { BREAK_LINE, LIST_ITEM } from "@/utils/textConst";

/**
 * Markdownのリストをパースする
 * @param content パース対象の文字列
 * @returns パース結果
 */
export const parseLists = (
  content: string
): MarkdownList[] => {

  const lines = content.split(BREAK_LINE);

  const result: MarkdownList[] = [];

  let currentItems: string[] = [];

  let startLine = 0;

  const flush = () => {
    if (currentItems.length === 0) {
      return;
    }

    result.push({
      items: [...currentItems],
      lineNumber: startLine
    });

    currentItems = [];
    startLine = 0;
  };

  lines.forEach((line, index) => {
    const match = line.match(LIST_ITEM);

    if (match) {

      if (currentItems.length === 0) {
        startLine = index + 1;
      }

      currentItems.push(match[2].trim());

      return;
    }

    flush();
  });

  flush();

  return result;
};

import { MarkdownTable } from "@/models/MarkdownTable";
import { BREAK_LINE, TABLE_SEPARATOR } from "@/utils/textConst";

/**
 * マークダウンコンテンツからテーブルを抽出する
 * @param content マークダウンコンテンツ
 * @returns 抽出されたテーブルの配列
 */
export const parseTables = (
  content: string
): MarkdownTable[] => {

  const lines = content.split(BREAK_LINE);
  const tables: MarkdownTable[] = [];

  let index = 0;
  while (
    index < lines.length
  ) {
    const headerLine = lines[index];
    const separatorLine = lines[index + 1];
    const isHeader = headerLine?.includes("|");

    const isSeparator =
      separatorLine &&
      TABLE_SEPARATOR.test(separatorLine);

    if (!isHeader || !isSeparator) {
      index++;
      continue;
    }

    const headers = parseTableRow(headerLine);
    const rows: string[][] = [];

    let rowIndex = index + 2;

    while (
      rowIndex < lines.length &&
      lines[rowIndex].includes("|")
    ) {
      rows.push(parseTableRow(lines[rowIndex]));
      rowIndex++;
    }

    tables.push({
      headers,
      rows,
      lineNumber: index + 1
    });

    index = rowIndex;
  }

  return tables;
};

/**
 * テーブルの行を解析する
 * @param line テーブルの行
 * @returns 解析されたセルの配列
 */
const parseTableRow = (
  line: string
): string[] => {

  return line
    .split("|")
    .map(
      item => item.trim()
    )
    .filter(
      item => item.length > 0
    );
};

import { Heading } from "@/models/Heading";
import { BREAK_LINE, HEADING } from "@/utils/textConst";

/**
 * マークダウンコンテンツから見出しを抽出する
 * @param content マークダウンコンテンツ
 * @returns 抽出された見出しの配列
 */
export const parseHeadings = (
  content: string
): Heading[] => {

  const lines = content.split(BREAK_LINE);

  return lines
    .map((line, index) => {
      const match = line.match(HEADING);

      if (!match) {
        return null;
      }

      return {
        level: match[1].length,
        text: match[2].trim(),
        lineNumber: index + 1
      };
    })
    .filter(
      (heading): heading is Heading => heading !== null
    );
};

/**
 * 見出しからタイトルを抽出する
 * @param headings 見出しの配列
 * @returns 抽出されたタイトル
 */
export const extractTitle = (
  headings: Heading[]
): string => {

  const h1 =
    headings.find(
      heading => heading.level === 1
    );

  return h1?.text ?? "";
};

import type { DocumentModel } from "../models/DocumentModel";
import type { Task } from "../models/Task";

const TASK_KEYWORDS = [

  "作成",
  "実装",
  "対応",
  "修正",
  "追加",
  "削除",
  "変更",

  "作る",
  "実施",

  "Create",
  "Implement",
  "Fix",
  "Update"

];

const TASK_SECTION_KEYWORDS = [

  "タスク",
  "Task",
  "TODO",
  "ToDo",
  "WBS",
  "実装"

];

export const extractTasks = (
  document: DocumentModel
): Task[] => {

  const tasksFromLists =
    extractTasksFromLists(
      document
    );

  const uniqueTasks =
    uniqueByTitle(
      tasksFromLists
    );

  return uniqueTasks.map(
    (task, index) => ({
      ...task,
      id:
        "TSK-" +
        String(
          index + 1
        ).padStart(
          3,
          "0"
        )
    })
  );
};

const extractTasksFromLists = (
  document: DocumentModel
): Omit<Task, "id">[] => {

  const tasks:
    Omit<Task, "id">[] = [];

  for (
    const list of document.lists
  ) {

    for (
      const item of list.items
    ) {

      if (
        !isTaskCandidate(
          item
        )
      ) {
        continue;
      }

      tasks.push({
        title:
          normalizeTitle(
            item
          ),

        description:
          item,

        status:
          "TODO",

        priority:
          "NORMAL",

        source:
          document.title
      });
    }
  }

  return tasks;
};

const isTaskCandidate = (
  text: string
): boolean => {

  return TASK_KEYWORDS.some(
    keyword =>
      text.includes(
        keyword
      )
  );
};

const normalizeTitle = (
  text: string
): string => {

  return text
    .replace(
      /^\[[ xX]\]\s*/,
      ""
    )
    .trim();
};

const uniqueByTitle = <
  T extends {
    title: string;
  }
>(
  values: T[]
): T[] => {

  const map =
    new Map<
      string,
      T
    >();

  values.forEach(
    value => {

      if (
        map.has(
          value.title
        )
      ) {
        return;
      }

      map.set(
        value.title,
        value
      );

    }
  );

  return [
    ...map.values()
  ];
};

export const hasTaskSection =
  (
    document: DocumentModel
  ): boolean => {

    return document.headings.some(
      heading =>

        TASK_SECTION_KEYWORDS.some(
          keyword =>
            heading.text.includes(
              keyword
            )
        )

    );
  };

import fs from "fs";
import path from "path";

const IGNORE_DIRS = [

  "node_modules",
  "dist",
  ".git",
  "coverage"

];

const collectFiles =
(
  directory: string
): string[] => {

  const entries =
    fs.readdirSync(
      directory,
      {
        withFileTypes:true
      }
    );

  return entries.flatMap(
    entry => {

      const fullPath =
        path.join(
          directory,
          entry.name
        );

      if (
        entry.isDirectory()
      ) {

        if (
          IGNORE_DIRS.includes(
            entry.name
          )
        ) {
          return [];
        }

        return collectFiles(
          fullPath
        );
      }

      return [fullPath];

    }
  );
};

const exportSource =
(
  targetDir: string,
  outputFile: string
) => {

  const files =
    collectFiles(
      targetDir
    );

  let output = "";

  for (
    const file of files
  ) {

    output +=
      "\n" +
      "=".repeat(80) +
      "\n";

    output +=
      "FILE: " +
      file +
      "\n";

    output +=
      "=".repeat(80) +
      "\n\n";

    output +=
      fs.readFileSync(
        file,
        "utf-8"
      );

    output +=
      "\n\n";
  }

  fs.writeFileSync(
    outputFile,
    output,
    "utf-8"
  );

  console.log(
    "Export Complete:"
    + outputFile
  );
};

const targetDir =
  process.argv[2];

const outputFile =
  process.argv[3] ??
  "source-export.txt";

if (
  !targetDir
) {

  console.error(
    "Usage: export-source <dir>"
  );

  process.exit(1);

}

exportSource(
  targetDir,
  outputFile
);
