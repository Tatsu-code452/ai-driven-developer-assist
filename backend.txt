import cors from "cors";
import express from "express";
import { errorHandler } from "./middlewares/errorHandler";
import routes from "./routes";

export const createApp = () => {

  const app = express();

  app.use(cors());

  app.use(
    express.json({
      limit: "10mb"
    })
  );

  app.use(
    express.urlencoded({
      extended: true
    })
  );

  app.get(
    "/",
    (_req, res) => {
      res.json({
        application: "AI Driven Dev Support",
        version: "0.1.0"
      });
    }
  );

  app.use("/api", routes);

  app.use(errorHandler);

  return app;
};

(
  req: Request,
  res: Response
) => {

  const items =
    loadContexts(
      "data/generated"
    );

  const item =
    items.find(
      context =>
        context.id ===

    );

      .json({

        success: false

      });

  }

  res.json({

    success: true,

    item

  });

};

  req: Request,
  res: Response

          "content is required"

      });

    success: true,

  });

};

import { Request, Response } from "express";

import {
  loadTasks
} from "@/repositories/TaskRepository";

import {
  loadDocuments
} from "@/repositories/DocumentRepository";

export const getDashboard =
(
  _req: Request,
  res: Response
) => {

  const tasks =
    loadTasks(
      "data/generated"
    );

  const documents =
    loadDocuments(
      "data/generated"
    );

  const completedTasks =
    tasks.filter(
      task =>
        task.status === "DONE"
    );

  return res.json({

    success: true,

    summary: {

      taskCount:
        tasks.length,

      completedTaskCount:
        completedTasks.length,

      documentCount:
        documents.length,

      progress:
        tasks.length === 0
          ? 0
          : Math.round(
              (
                completedTasks.length /
                tasks.length
              ) * 100
            )

    },

    latestTasks:
      tasks.slice(-5),

    latestDocuments:
      documents.slice(-5)

  });

};

import fs from "fs";
import path from "path";

import type {
  Context
} from "@/models/Context";

const FILE_NAME =
  "contexts.json";

export const loadContexts =
(
  outputDir: string
): Context[] => {

  const filePath =
    path.join(
      outputDir,
      FILE_NAME
    );

  if (
    !fs.existsSync(
      filePath
    )
  ) {
    return [];
  }

  return JSON.parse(
    fs.readFileSync(
      filePath,
      "utf-8"
    )
  );

};

export const saveContexts =
(
  items: Context[],
  outputDir: string
) => {

  fs.mkdirSync(
    outputDir,
    {
      recursive: true
    }
  );

  const filePath =
    path.join(
      outputDir,
      FILE_NAME
    );

  fs.writeFileSync(
    filePath,
    JSON.stringify(
      items,
      null,
      2
    ),
    "utf-8"
  );

  return filePath;

};

  const filePath =

  ) {
    return [];
  }

export const saveDocuments =
(
  documents: Document[],
  outputDir: string
) => {

  fs.mkdirSync(
    outputDir,
    {
      recursive:true
    }
  );

  const filePath =
    path.join(
      outputDir,
      FILE_NAME
    );

  fs.writeFileSync(
    filePath,
    JSON.stringify(
      documents,
      null,
      2
    ),
    "utf-8"
  );

  return filePath;
};

export const mergeDocuments =
(
  current: Document[],
  incoming: Document[]
): Document[] => {

  const map =
    new Map<
      string,
      Document
    >();

  current.forEach(
    document =>

      map.set(
        document.id,
        document
      )

  );

  incoming.forEach(
    document => {

      const exists =
        [...map.values()]
          .find(
            item =>
              item.title ===
              document.title
          );

      if (
        exists
      ) {
        return;
      }

      map.set(
        document.id,
        document
      );

    }
  );

  return [
    ...map.values()
  ];

};

import path from "path";

import {
  loadTasks,
  mergeTasks,
  saveTasks
} from "@/repositories/TaskRepository";

import {
  loadDocuments,
  mergeDocuments,
  saveDocuments
} from "@/repositories/DocumentRepository";

import {
  createIdAllocator
} from "@/services/idGenerator/IdGenerator";

import {
  extractTasks
} from "@/services/TaskExtractor";

import {
  detectDocumentType,
  extractDocuments
} from "@/services/DocumentExtractor";

import {
  parseMarkdownContent
} from "@/services/markdownParser/parseMarkdownContent";

import {
  loadContexts,
  saveContexts
} from "@/repositories/ContextRepository";
import { buildContext } from "@/services/contextBuilder/buildContext";

export const convertMarkdown =
  (
    markdown: string,
    outputRoot = "data"
  ) => {

    const document =
      parseMarkdownContent(
        markdown
      );

    const outputDir =
      path.join(
        outputRoot,
        "generated"
      );

    const currentTasks =
      loadTasks(
        outputDir
      );

    const currentDocuments =
      loadDocuments(
        outputDir
      );

    const taskAllocator =
      createIdAllocator(
        "TSK",
        currentTasks
      );

    const docType =
      detectDocumentType(
        document
      );

    const documentAllocator =
      createIdAllocator(
        docType,
        currentDocuments
      );

    const newTasks =
      extractTasks(
        document,
        {
          nextId:
            taskAllocator
        }
      );

    const newDocuments =
      extractDocuments(
        document,
        {
          nextId:
            documentAllocator
        }
      );

    const mergedTasks =
      mergeTasks(
        currentTasks,
        newTasks
      );

    const mergedDocuments =
      mergeDocuments(
        currentDocuments,
        newDocuments
      );

    const tasksFile =
      saveTasks(
        mergedTasks,
        outputDir
      );

    const documentsFile =
      saveDocuments(
        mergedDocuments,
        outputDir
      );

    const contexts =
      loadContexts(
        outputDir
      );

    const contextAllocator =
      createIdAllocator(
        "CTX",
        contexts
      );

    const context =
      buildContext(
        contextAllocator(),
        document.title,
        docType,
        newTasks.map(
          task => task.id
        ),
        newDocuments.map(
          doc => doc.id
        )
      );

    contexts.push(
      context
    );

    saveContexts(
      contexts,
      outputDir
    );

    return {

      document,

      tasks:
        mergedTasks,

      documents:
        mergedDocuments,

      output: {

        tasksFile,

        documentsFile

      },

      summary: {

        taskCount:
          mergedTasks.length,

        documentCount:
          mergedDocuments.length

      }

    };
  };

import type { DocumentModel } from "../models/DocumentModel";

type Options = {

export const extractDocuments = (
  document: DocumentModel,
  options?: Options
): Document[] => {

  const type =
    detectDocumentType(
      document
    );

  return [

    {
      id:
        options?.nextId
          ? options.nextId()

  (

  ) => {

    const title =
      document.title
        .toLowerCase();

    const fileName =
      document.fileName
        .toLowerCase();

      title + " " +

      )

      return "TST";
    }

    if (

      )
    ) {

  };

type Entity = {
  id: string;
};

export const getNextId = (
  prefix: string,
  items: Entity[]
): string => {

  const maxNo =
    items
      .filter(
        item =>
          item.id.startsWith(
            prefix + "-"
          )
      )
      .map(
        item =>
          Number(
            item.id
              .replace(
                prefix + "-",
                ""
              )
          )
      )
      .filter(
        value =>
          !Number.isNaN(
            value
          )
      )
      .reduce(
        (
          current,
          value
        ) =>
          Math.max(
            current,
            value
          ),
        0
      );

  return (
    prefix +
    "-" +
    String(
      maxNo + 1
    ).padStart(
      3,
      "0"
    )
  );
};

export const createIdAllocator =
(
  prefix: string,
  items: Entity[]
) => {

  let current =
    Number(
      getNextId(
        prefix,
        items
      ).replace(
        prefix + "-",
        ""
      )
    );

  return () => {

    const value =
      prefix +
      "-" +
      String(
        current
      ).padStart(
        3,
        "0"
      );

    current++;

    return value;

  };

};

import { MarkdownList } from "@/models/MarkdownList";
import { BREAK_LINE, LIST_ITEM } from "@/utils/textConst";

/**
 * Markdownのリストをパースする
 * @param content パース対象の文字列
 * @returns パース結果
 */
export const parseLists = (
  content: string
): MarkdownList[] => {

  const lines = content.split(BREAK_LINE);

  const result: MarkdownList[] = [];

  let currentItems: string[] = [];

  let startLine = 0;

  const flush = () => {
    if (currentItems.length === 0) {
      return;
    }

    result.push({
      items: [...currentItems],
      lineNumber: startLine
    });

    currentItems = [];
    startLine = 0;
  };

  lines.forEach((line, index) => {
    const match = line.match(LIST_ITEM);

    if (match) {

      if (currentItems.length === 0) {
        startLine = index + 1;
      }

      currentItems.push(match[2].trim());

      return;
    }

    flush();
  });

  flush();

  return result;
};

import { MarkdownTable } from "@/models/MarkdownTable";
import { BREAK_LINE, TABLE_SEPARATOR } from "@/utils/textConst";

/**
 * マークダウンコンテンツからテーブルを抽出する
 * @param content マークダウンコンテンツ
 * @returns 抽出されたテーブルの配列
 */
export const parseTables = (
  content: string
): MarkdownTable[] => {

  const lines = content.split(BREAK_LINE);
  const tables: MarkdownTable[] = [];

  let index = 0;
  while (
    index < lines.length
  ) {
    const headerLine = lines[index];
    const separatorLine = lines[index + 1];
    const isHeader = headerLine?.includes("|");

    const isSeparator =
      separatorLine &&
      TABLE_SEPARATOR.test(separatorLine);

    if (!isHeader || !isSeparator) {
      index++;
      continue;
    }

    const headers = parseTableRow(headerLine);
    const rows: string[][] = [];

    let rowIndex = index + 2;

    while (
      rowIndex < lines.length &&
      lines[rowIndex].includes("|")
    ) {
      rows.push(parseTableRow(lines[rowIndex]));
      rowIndex++;
    }

    tables.push({
      headers,
      rows,
      lineNumber: index + 1
    });

    index = rowIndex;
  }

  return tables;
};

/**
 * テーブルの行を解析する
 * @param line テーブルの行
 * @returns 解析されたセルの配列
 */
const parseTableRow = (
  line: string
): string[] => {

  return line
    .split("|")
    .map(
      item => item.trim()
    )
    .filter(
      item => item.length > 0
    );
};

import { Heading } from "@/models/Heading";

/**
 * マークダウンコンテンツから見出しを抽出する
 * @param content マークダウンコンテンツ
 * @returns 抽出された見出しの配列
 */
export const parseHeadings = (
  content: string
): Heading[] => {

  const lines = content.split(BREAK_LINE);

  return lines
    .map((line, index) => {
      const match = line.match(HEADING);

      if (!match) {
        return null;
      }

      return {
        level: match[1].length,
        text: match[2].trim(),
        lineNumber: index + 1
      };
    })
    .filter(
      (heading): heading is Heading => heading !== null
    );
};

/**
 * 見出しからタイトルを抽出する
 * @param headings 見出しの配列
 * @returns 抽出されたタイトル
 */
export const extractTitle = (
  headings: Heading[]
): string => {

  const h1 =
    headings.find(
      heading => heading.level === 1
    );

  return h1?.text ?? "";
};

const TASK_KEYWORDS = [

  "実装",
  "対応",
  "修正",
  "追加",
  "削除",
  "変更",

  "作る",
  "実施",

  "Create",
  "Implement",
  "Fix",
  "Update"

];

const TASK_SECTION_KEYWORDS = [

  "タスク",
  "Task",
  "TODO",
  "ToDo",
  "WBS",
  "実装"

];

export const extractTasks = (
  document: DocumentModel,
  options?: Options
): Task[] => {

  const tasks =
    extractTasksFromLists(
      document
    );

  const uniqueTasks =
    uniqueByTitle(
      tasks
    );

  return uniqueTasks.map(
    task => ({

      ...task,

      id:
        options?.nextId
          ? options.nextId()
          : "TSK-000"

    })
  );

};

const extractTasksFromLists = (
  document: DocumentModel
): Omit<Task, "id">[] => {

  const tasks:
    Omit<Task, "id">[] = [];

  for (
    const list of document.lists
  ) {

    for (
      const item of list.items
    ) {

      if (
        !isTaskCandidate(
          item
        )
      ) {
        continue;
      }

      tasks.push({
        title:
          normalizeTitle(
            item
          ),

        description:
          item,

        status:
          "TODO",

        priority:
          "NORMAL",

        source:
          document.title
      });
    }
  }

  return tasks;
};

const isTaskCandidate = (
  text: string
): boolean => {

  return TASK_KEYWORDS.some(
    keyword =>
      text.includes(
        keyword
      )
  );
};

const normalizeTitle = (
  text: string
): string => {

  return text
    .replace(
      /^\[[ xX]\]\s*/,
      ""
    )
    .trim();
};

const uniqueByTitle = <
  T extends {
    title: string;
  }
>(
  values: T[]
): T[] => {

  const map =
    new Map<
      string,
      T
    >();

  values.forEach(
    value => {

      if (
        map.has(
          value.title
        )
      ) {
        return;
      }

      map.set(
        value.title,
        value
      );

    }
  );

  return [
    ...map.values()
  ];
};

export const hasTaskSection =
  (
    document: DocumentModel
  ): boolean => {

    return document.headings.some(
      heading =>

        TASK_SECTION_KEYWORDS.some(
          keyword =>
            heading.text.includes(
              keyword
            )
        )

    );
  };

import fs from "fs";
import path from "path";

const IGNORE_DIRS = [

  "node_modules",
  "dist",
  ".git",
  "coverage"

];

const collectFiles =
(
  directory: string
): string[] => {

  const entries =
    fs.readdirSync(
      directory,
      {
        withFileTypes:true
      }
    );

  return entries.flatMap(
    entry => {

      const fullPath =
        path.join(
          directory,
          entry.name
        );

      if (
        entry.isDirectory()
      ) {

        if (
          IGNORE_DIRS.includes(
            entry.name
          )
        ) {
          return [];
        }

        return collectFiles(
          fullPath
        );
      }

      return [fullPath];

    }
  );
};

const exportSource =
(
  targetDir: string,
  outputFile: string
) => {

  const files =
    collectFiles(
      targetDir
    );

  let output = "";

  for (
    const file of files
  ) {

    output +=
      "\n" +
      "=".repeat(80) +
      "\n";

    output +=
      "FILE: " +
      file +
      "\n";

    output +=
      "=".repeat(80) +
      "\n\n";

    output +=
      fs.readFileSync(
        file,
        "utf-8"
      );

    output +=
      "\n\n";
  }

  fs.writeFileSync(
    outputFile,
    output,
    "utf-8"
  );

  console.log(
    "Export Complete:"
    + outputFile
  );
};

const targetDir =
  process.argv[2];

const outputFile =
  process.argv[3] ??
  "source-export.txt";

if (
  !targetDir
) {

  console.error(
    "Usage: export-source <dir>"
  );

  process.exit(1);

}

exportSource(
  targetDir,
  outputFile
);
