const resolveRootPath =
(
  req: Request
) => {

  const projectId =
    String(
      req.query.projectId ??
      ""
    );

  if (
    !projectId
  ) {

    return "data/generated";

  }

  const project =
    loadProjectById(
      projectId,
      DATA_DIRECTORY,
    );

  if (
    !project?.rootPath
  ) {

    return "data/generated";

  }

  return project.rootPath;

};

import { createProjectPageResponse, createProjectResponseOrNotFound, getFilteredProjects } from "@/controllers/project/projectControllerHelper";

export const getProjects: ApiMethod = (
  req: Request,
  res: Response,

  const context: ValidationContext = {};

import type { Response } from "express";

import { toProjectResponse } from "@/controllers/project/projectMapper";
import { DATA_DIRECTORY, PAGE_SIZE } from "@/controllers/project/projectTypes";
import { ProjectRecord } from "@/models/Project";
import { loadDocumentsGrouped } from "@/repositories/DocumentRepository";
import {
  loadProjects
} from "@/repositories/ProjectRepository";
import { loadTasksGrouped } from "@/repositories/TaskRepository";
import {
  createNotFoundResponse,
  createPagedResponse,
  createSuccessResponse
} from "@/utils/responseUtils";


export const loadProjectRelations = () => ({
  tasks: loadTasksGrouped(DATA_DIRECTORY),
  documents: loadDocumentsGrouped(DATA_DIRECTORY),
});

export const createProjectPageResponse = (
  res: Response,
  requestedPage: number,
  projects: ProjectRecord[]
) => {
  const relations = loadProjectRelations();

  const { page, start } = getPageInfo(projects.length, PAGE_SIZE, requestedPage);

  const items = projects
    .slice(
      start,
      start + PAGE_SIZE,
    )
    .map(project =>
      toProjectResponse(
        project,
        relations.tasks,
        relations.documents,
      ),
    );

  return createPagedResponse(
    res,
    {
      items,
      total: projects.length,
      page,
      pageSize: PAGE_SIZE,
    },
  );
};

export const createProjectResponse = (
  res: Response,
  project: ProjectRecord
) => {
  const relations =
    loadProjectRelations();

  return createSuccessResponse(
    res,
    toProjectResponse(
      project,
      relations.tasks,
      relations.documents,
    ),
  );
};

export const createProjectResponseOrNotFound = (
  res: Response,
  project: ProjectRecord | undefined,
) => {
  if (!project) {
    return createNotFoundResponse(
      res,
      "プロジェクトが見つかりません",
    );
  }

  return createProjectResponse(res, project);
};

export const getFilteredProjects = (keyword: string, statusFilter: string) => {
  const projects =
    loadProjects(DATA_DIRECTORY)
      .filter(project => {

        const matchesKeyword =
          !keyword ||
          project.id
            .toLowerCase()
            .includes(keyword) ||
          project.name
            .toLowerCase()
            .includes(keyword);

        const matchesStatus =
          !statusFilter ||
          project.status
            .toLowerCase()
            .includes(statusFilter);

        return (
          matchesKeyword &&
          matchesStatus
        );
      })
      .sort(
        (left, right) =>
          left.name.localeCompare(right.name) ||
          left.id.localeCompare(right.id),
      );

  return projects;
};


export const getPageInfo = (totalItems: number, pageSize: number, requestedPage: number) => {
  const pageCount = Math.ceil(totalItems / pageSize);
  const page = pageCount === 0 ? 1 : Math.min(requestedPage, pageCount);
  const start = (page - 1) * pageSize;
  return { page, pageCount, start };
};

import type { Document } from "@/models/Document";
import type {
    ProjectInput,
    ProjectRecord,
} from "@/models/Project";
import type { Task } from "@/models/Task";
import { isStringProperty } from "@/utils/propUtils";
import { isObject } from "@/utils/typeUtils";

export const isProjectInput = (
    value: unknown,
): value is ProjectInput =>
    isObject(value) &&
    isStringProperty(value, "name") &&
    (
        value.description === undefined ||
        isStringProperty(value, "description")
    ) &&
    (
        value.status === undefined ||
        isStringProperty(value, "status")
    ) &&
    (
        value.rootPath === undefined ||
        isStringProperty(value, "rootPath")
    );

export const projectInputFrom = (
    body: unknown,
): ProjectInput | undefined => {
    if (!isProjectInput(body)) {
        return undefined;
    }

    const name = body.name.trim();

    if (!name) {
        return undefined;
    }

    return {
        name,
        description: body.description ?? "",
        status: body.status ?? "",
        rootPath: body.rootPath ?? "",
    };
};

export const toProjectResponse = (
    project: ProjectRecord,
    taskMap: Map<string, Task[]>,
    documentMap: Map<string, Document[]>,
) => ({
    ...project,
    taskCount:
        taskMap.get(project.id)?.length ?? 0,
    documentCount:
        documentMap.get(project.id)?.length ?? 0,
});

import type { Request, Response } from "express";

import { projectInputFrom } from "@/controllers/project/projectMapper";
import type { ProjectInput } from "@/models/Project";
import { convertToPositiveIntegerOrDefault } from "@/utils/numberUtils";
import { createBadRequestResponse } from "@/utils/responseUtils";
import { isString } from "@/utils/typeUtils";

type QueryParams = {
    q?: string;
    status?: string;
};

export type ValidationContext = {
    id?: string;
    input?: ProjectInput;
    query?: QueryParams;
    requestedPage?: number;
};

export type Validator = (
    req: Request,
    res: Response,
    context: ValidationContext,
) => Response | undefined;

export const validate = (
    req: Request,
    res: Response,
    context: ValidationContext,
    validators: Validator[],
): Response | undefined => {
    for (const validator of validators) {
        const result = validator(
            req,
            res,
            context,
        );

        if (result) {
            return result;
        }
    }

    return undefined;
};

export const validateProjectId: Validator = (
    req,
    res,
    context,
) => {
    const id = req.params.id;

    if (!isString(id)) {
        return createBadRequestResponse(
            res,
            "プロジェクトIDを指定してください",
        );
    }

    context.id = id;
};

export const validateProjectInput: Validator = (
    req,
    res,
    context,
) => {
    const input =
        projectInputFrom(req.body);

    if (!input) {
        return createBadRequestResponse(
            res,
            "名称は必須です。説明と状態は文字列で指定してください",
        );
    }

    context.input = input;
};

export const hasProjectId = (
    context: ValidationContext,
): context is ValidationContext & {
    id: string;
} =>
    context.id !== undefined;

export const hasProjectInput = (
    context: ValidationContext,
): context is ValidationContext & {
    input: ProjectInput;
} =>
    context.input !== undefined;

export const validatePageQueryParam: Validator = (
    req: Request,
    res: Response,
    context: ValidationContext,
) => {
    const requestedPage = convertToPositiveIntegerOrDefault(req.query.page, 1);
    if (requestedPage === undefined) {
        return createBadRequestResponse(
            res,
            "ページ番号は正の整数で指定してください",
        );
    }

    context.requestedPage = requestedPage;

    if (
        (req.query.q !== undefined &&
            !isString(req.query.q)) ||
        (req.query.status !== undefined &&
            !isString(req.query.status))
    ) {
        return createBadRequestResponse(
            res,
            "検索条件は文字列で指定してください",
        );
    }

    context.query = {
        q: req.query.q ?? "",
        status: req.query.status ?? ""
    };
}

import { Request, Response } from "express";

import type {
  TaskInput,
  TaskPriority,
  TaskStatus,
} from "@/models/Task";
import {
  createTask,
  findTask,
  loadTasks,
  listTasks,
  updateTask,
} from "@/repositories/TaskRepository";

const DATA_DIR = "data/generated";
const PROJECT_ID_PATTERN = /^PRJ-\d{3}$/;
const TASK_STATUSES: TaskStatus[] = ["TODO", "DOING", "DONE"];
const TASK_PRIORITIES: TaskPriority[] = ["LOW", "NORMAL", "HIGH"];
const ALLOWED_TRANSITIONS: Record<TaskStatus, TaskStatus[]> = {
  TODO: ["TODO", "DOING"],
  DOING: ["DOING", "DONE"],
  DONE: ["DONE", "DOING"],
};

const isString = (value: unknown): value is string =>
  typeof value === "string";

const isOneOf = <T extends string>(
  value: unknown,
  allowed: T[],
): value is T =>
  isString(value) && allowed.some((item) => item === value);

const errorResponse = (
  res: Response,
  status: number,
  message: string,
) =>
  res.status(status).json({
    success: false,
    message,
  });

const parseTaskInput = (
  body: unknown,
): TaskInput | string => {
  if (!body || typeof body !== "object" || Array.isArray(body)) {
    return "入力内容が不正です。";
  }

  const input = body as Record<string, unknown>;
  const title = isString(input.title) ? input.title.trim() : "";
  const description =
    input.description === undefined
      ? ""
      : isString(input.description)
        ? input.description
        : undefined;
  if (!title) return "タイトルを入力してください。";
  if (title.length > 200) return "タイトルは200文字以内で入力してください。";
  if (description === undefined) return "説明は文字列で入力してください。";
  if (!isOneOf(input.status, TASK_STATUSES)) {
    return "状態が不正です。";
  }
  if (!isOneOf(input.priority, TASK_PRIORITIES)) {
    return "優先度が不正です。";
  }
  if (
    !isString(input.projectId) ||
    !PROJECT_ID_PATTERN.test(input.projectId)
  ) {
    return "プロジェクトを選択してください。";
  }

  return {
    title,
    description,
    status: input.status,
    priority: input.priority,
    projectId: input.projectId,
  };
};

export const getTasks = (
  req: Request,
  res: Response,
) => {
  const { q = "", projectId, status, priority, page = "1" } = req.query;
  if (
    !isString(q) ||
    (projectId !== undefined && !isString(projectId)) ||
    (status !== undefined && !isString(status)) ||
    (priority !== undefined && !isString(priority)) ||
    !isString(page) ||
    !/^[1-9]\d*$/.test(page)
  ) {
    return errorResponse(res, 400, "検索条件が不正です。");
  }
  if (
    projectId !== undefined &&
    projectId !== "unassigned" &&
    !PROJECT_ID_PATTERN.test(projectId)
  ) {
    return errorResponse(res, 400, "プロジェクトIDが不正です。");
  }
  if (status !== undefined && !isOneOf(status, TASK_STATUSES)) {
    return errorResponse(res, 400, "状態が不正です。");
  }
  if (priority !== undefined && !isOneOf(priority, TASK_PRIORITIES)) {
    return errorResponse(res, 400, "優先度が不正です。");
  }

  const result = listTasks(loadTasks(DATA_DIR), {
    q,
    projectId,
    status,
    priority,
    page: Number(page),
  });
  return res.json({
    success: true,
    ...result,
  });
};

export const getTaskDetail = (
  req: Request,
  res: Response,
) => {
  const id = req.params.id;
  if (!isString(id)) return errorResponse(res, 400, "タスクIDが不正です。");
  const task = findTask(id, DATA_DIR);
  if (!task) return errorResponse(res, 404, "タスクが見つかりません。");
  return res.json({ success: true, item: task });
};

export const postTask = (
  req: Request,
  res: Response,
) => {
  const input = parseTaskInput(req.body);
  if (typeof input === "string") {
    return errorResponse(res, 400, input);
  }
  if (input.status !== "TODO") {
    return errorResponse(res, 400, "新規タスクの状態はTODOで登録してください。");
  }

  const task = createTask(input, DATA_DIR);
  return res.status(201).json({ success: true, item: task });
};

export const putTask = (
  req: Request,
  res: Response,
) => {
  const id = req.params.id;
  if (!isString(id)) return errorResponse(res, 400, "タスクIDが不正です。");

  const current = findTask(id, DATA_DIR);
  if (!current) return errorResponse(res, 404, "タスクが見つかりません。");

  const input = parseTaskInput(req.body);
  if (typeof input === "string") {
    return errorResponse(res, 400, input);
  }
  if (!ALLOWED_TRANSITIONS[current.status]?.includes(input.status)) {
    return errorResponse(res, 400, "許可されていない状態遷移です。");
  }

  const task = updateTask(id, input, DATA_DIR);
  if (!task) return errorResponse(res, 404, "タスクが見つかりません。");
  return res.json({ success: true, item: task });
};

export const loadDocumentsGrouped =
  (
    outputDir: string
  ): Map<string, Document[]> => {

    const documents =
      loadDocuments(
        outputDir
      );

    return documents.reduce(
      (
        acc,
        document
      ) => {
        if (!document.projectId) {
          return acc;
        }

        const projectDocuments = acc.get(document.projectId) ?? [];
        projectDocuments.push(document);
        acc.set(document.projectId, projectDocuments);

        return acc;
      },
      new Map<string, Document[]>()
    );
  }

export const saveDocuments =
  (
    documents: Document[],
    outputDir: string
  ) => {

    fs.mkdirSync(

        recursive: true
      }
    );

    const filePath =
      path.join(

        FILE_NAME
      );

export const mergeDocuments =
  (
    current: Document[],
    incoming: Document[]
  ): Document[] => {

    const map =
      new Map<
        string,
        Document
      >();

    current.forEach(
      document =>

        map.set(
          document.id,
          document
        )

    );

    incoming.forEach(
      document => {

        const exists =
          [...map.values()]
            .find(
              item =>
                item.title ===
                document.title
            );

        if (
          exists
        ) {
          return;
        }

        map.set(
          document.id,
          document
        );

      }
    );

    return [
      ...map.values()
    ];

  };

import fs from "fs";
import path from "path";

import type {
  ManagementDataIndex,
  ManagementDataIndexStatus,
  ManagementDataItem,
} from "@/models/ManagementData";
import {
  MANAGEMENT_DATA_SECTION_TITLE,
  parseManagementDataLinks,
} from "@/services/managementDataIndex/parseManagementDataIndex";

const README_FILE_NAME = "README.md";
const EXTERNAL_LINK_PATTERN = /^[a-z][a-z0-9+.-]*:/i;

const isDirectory = (target: string): boolean =>
  fs.existsSync(target) && fs.statSync(target).isDirectory();

const isFile = (target: string): boolean =>
  fs.existsSync(target) && fs.statSync(target).isFile();

const createIndex = (
  status: ManagementDataIndexStatus,
  message: string,
  rootPath: string,
  items: ManagementDataItem[] = [],
): ManagementDataIndex => ({ status, message, rootPath, items });

/**
 * リンク先を管理データパス配下に限定して解決する。
 * 外部URL・絶対パス・配下の外へ出る相対パスは対象外(undefined)とする。
 */
const resolveInsideRoot = (
  root: string,
  href: string,
): { absolute: string; relative: string } | undefined => {
  if (EXTERNAL_LINK_PATTERN.test(href) || path.isAbsolute(href)) {
    return undefined;
  }

  const absolute = path.resolve(root, href);
  const relative = path.relative(root, absolute);

  if (
    !relative ||
    relative.startsWith("..") ||
    path.isAbsolute(relative)
  ) {
    return undefined;
  }

  return { absolute, relative: relative.split(path.sep).join("/") };
};

export const loadManagementDataIndex = (
  rootPath: string,
): ManagementDataIndex => {
  const configuredPath = rootPath.trim();

  if (!configuredPath) {
    return createIndex(
      "root-path-not-set",
      "プロジェクトに管理データパスが設定されていません。",
      rootPath,
    );
  }

  const root = path.resolve(configuredPath);

  if (!isDirectory(root)) {
    return createIndex(
      "root-path-not-found",
      "管理データパスのフォルダーが見つかりません。パスとバックエンドの実行環境を確認してください。",
      rootPath,
    );
  }

  const readmePath = path.join(root, README_FILE_NAME);

  if (!isFile(readmePath)) {
    return createIndex(
      "readme-not-found",
      `管理データパス直下に${README_FILE_NAME}がありません。`,
      rootPath,
    );
  }

  const links = parseManagementDataLinks(fs.readFileSync(readmePath, "utf-8"));

  if (!links) {
    return createIndex(
      "section-not-found",
      `${README_FILE_NAME}に「${MANAGEMENT_DATA_SECTION_TITLE}」の見出しがありません。`,
      rootPath,
    );
  }

  const items = links.flatMap((link): ManagementDataItem[] => {
    const target = resolveInsideRoot(root, link.href);

    return target
      ? [{
          title: link.title,
          path: target.relative,
          exists: fs.existsSync(target.absolute),
        }]
      : [];
  });

  return createIndex("ok", "", rootPath, items);
};

import fs from "fs";
import path from "path";

import type { ProjectInput, ProjectRecord } from "@/models/Project";
import { getNextId } from "@/services/idGenerator/IdGenerator";
import { isStringProperty } from "@/utils/propUtils";
import { isObject } from "@/utils/typeUtils";

const FILE_NAME = "projects.json";

const INITIAL_PROJECT: ProjectRecord = {
  id: "PRJ-001",
  name: "AI駆動型開発支援システム",
  description: "",
  status: "進行中",
  rootPath: "",
};

const isProjectRecord = (value: unknown): value is ProjectRecord =>
  isObject(value) &&
  isStringProperty(value, "id") &&
  isStringProperty(value, "name") &&
  isStringProperty(value, "description") &&
  isStringProperty(value, "status") &&
  isStringProperty(value, "rootPath");


export const loadProjects = (outputDir: string): ProjectRecord[] => {
  const filePath = path.join(outputDir, FILE_NAME);

  if (!fs.existsSync(filePath)) {
    return [{ ...INITIAL_PROJECT }];
  }

  const parsed: unknown = JSON.parse(fs.readFileSync(filePath, "utf-8"));
  if (!Array.isArray(parsed) || !parsed.every(isProjectRecord)) {
    throw new Error("プロジェクトデータの形式が不正です");
  }

  return parsed as ProjectRecord[];
};

export const loadProjectById = (
  id: string,
  outputDir: string,
): ProjectRecord | undefined => {
  const projects = loadProjects(outputDir);
  return projects.find((project) => project.id === id);
};

export const saveProjects = (
  projects: ProjectRecord[],
  outputDir: string,
): string => {
  fs.mkdirSync(outputDir, { recursive: true });

  const filePath = path.join(outputDir, FILE_NAME);
  const temporaryPath = `${filePath}.tmp`;
  fs.writeFileSync(temporaryPath, JSON.stringify(projects, null, 2), "utf-8");
  fs.renameSync(temporaryPath, filePath);

  return filePath;
};

export const createProject = (
  input: ProjectInput,
  outputDir: string,
): ProjectRecord => {
  const projects = loadProjects(outputDir);
  const project: ProjectRecord = {
    ...input,
    id: getNextId("PRJ", projects),
  };

  if (projects.some((item) => item.id === project.id)) {
    throw new Error(`プロジェクトIDが重複しています: ${project.id}`);
  }

  saveProjects([...projects, project], outputDir);
  return project;
};

export const updateProject = (
  id: string,
  input: ProjectInput,
  outputDir: string,
): ProjectRecord | undefined => {
  const projects = loadProjects(outputDir);
  const index = projects.findIndex((project) => project.id === id);

  if (index === -1) {
    return undefined;
  }

  const updated: ProjectRecord = { ...input, id };
  projects[index] = updated;
  saveProjects(projects, outputDir);

  return updated;
};

  page: number;
  pageSize: number;
  totalPages: number;
};

const isTask = (value: unknown): value is Task => {
  if (!value || typeof value !== "object" || Array.isArray(value)) {
    return false;
  }

  const task = value as Record<string, unknown>;
  return (
    typeof task.id === "string" &&
    typeof task.title === "string" &&
    typeof task.description === "string" &&
    taskStatuses.some((status) => status === task.status) &&
    taskPriorities.some((priority) => priority === task.priority) &&
    typeof task.source === "string" &&
    (task.projectId === undefined || typeof task.projectId === "string")
  );
};

export const loadTasks = (
  outputDir: string
): Task[] => {

  const filePath =
    path.join(
      outputDir,
      FILE_NAME
    );

  if (
    !fs.existsSync(
      filePath
    )
  ) {
    return [];
  }

  const parsed: unknown = JSON.parse(
    fs.readFileSync(
      filePath,
      "utf-8"
    )
  );

  if (!Array.isArray(parsed) || !parsed.every(isTask)) {
    throw new Error(
      `Invalid task data in ${filePath}`
    );
  }

  return parsed;
};

export const loadTasksGrouped = (
  outputDir: string
): Map<string, Task[]> => {
  const tasks = loadTasks(outputDir);
  return tasks.reduce(
    (acc, task) => {
      if (!task.projectId) {
        return acc;
      }

      const projectTasks = acc.get(task.projectId) ?? [];
      projectTasks.push(task);
      acc.set(task.projectId, projectTasks);

      return acc;
    },
    new Map<string, Task[]>()
  );
};

export const saveTasks = (
  tasks: Task[],
  outputDir: string
) => {

  fs.mkdirSync(
    outputDir,
    {
      recursive: true
    }
  );

  const filePath =
    path.join(
      outputDir,
      FILE_NAME
    );

  fs.writeFileSync(
    filePath,
    JSON.stringify(
      tasks,
      null,
      2
    ),
    "utf-8"
  );

  return filePath;
};

export const mergeTasks = (
  current: Task[],
  incoming: Task[]
): Task[] => {

  const map =
    new Map<
      string,
      Task
    >();

  current.forEach(
    task =>
      map.set(
        task.id,
        task
      )
  );

  incoming.forEach(
    task => {

      const exists =
        [...map.values()]
          .find(
            item =>
              item.title ===
              task.title
          );

      if (
        exists
      ) {
        return;
      }

      map.set(
        task.id,
        task
      );

    }
  );

  return [
    ...map.values()
  ];
};

export const listTasks = (
  tasks: Task[],
  filters: TaskListFilters
): TaskListResult => {
  const query = filters.q.trim().toLocaleLowerCase();
  const filtered = tasks
    .filter((task) => {
      if (
        filters.projectId === "unassigned" &&
        task.projectId
      ) {
        return false;
      }

      if (
        filters.projectId &&
        filters.projectId !== "unassigned" &&
        task.projectId !== filters.projectId
      ) {
        return false;
      }

      if (
        filters.status &&
        task.status !== filters.status
      ) {
        return false;
      }

      if (
        filters.priority &&
        task.priority !== filters.priority
      ) {
        return false;
      }

      if (
        query &&
        !`${task.id} ${task.title}`
          .toLocaleLowerCase()
          .includes(query)
      ) {
        return false;
      }

      return true;
    })
    .sort((left, right) => left.id.localeCompare(right.id));

  const totalPages = Math.max(
    1,
    Math.ceil(filtered.length / PAGE_SIZE)
  );
  const page = Math.min(filters.page, totalPages);
  const start = (page - 1) * PAGE_SIZE;

  return {
    items: filtered.slice(start, start + PAGE_SIZE),
    count: filtered.length,
    page,
    pageSize: PAGE_SIZE,
    totalPages,
  };
};

export const createTask = (
  input: TaskInput,
  outputDir: string
): Task => {
  const tasks = loadTasks(outputDir);
  const id = getNextId("TSK", [
    ...tasks,
    { id: RESERVED_TASK_ID },
  ]);
  const task: Task = {
    ...input,
    id,
    source: "手動登録",
  };

  saveTasks([...tasks, task], outputDir);
  return task;
};

export const findTask = (
  id: string,
  outputDir: string
): Task | undefined =>
  loadTasks(outputDir).find((task) => task.id === id);

export const updateTask = (
  id: string,
  input: TaskInput,
  outputDir: string
): Task | undefined => {
  const tasks = loadTasks(outputDir);
  const index = tasks.findIndex((task) => task.id === id);
  if (index < 0) return undefined;

  const updated: Task = {
    ...tasks[index],
    ...input,
    id,
  };
  tasks[index] = updated;
  saveTasks(tasks, outputDir);
  return updated;
};

  const worktrees: RepositoryState["worktrees"] = [];
  let worktree: RepositoryState["worktrees"][number] | undefined;

  for (const line of worktreeOutput.split(/\r?\n/)) {
    if (line.startsWith("worktree ")) {
      if (worktree) {
        worktrees.push(worktree);
      }

      const worktreePath = path.resolve(line.slice("worktree ".length));
      worktree = {
        path: worktreePath,
        branch: "",
        current: path.resolve(worktreePath) === path.resolve(root)
      };
    } else if (line.startsWith("branch ") && worktree) {
      worktree.branch = line
        .slice("branch ".length)
        .replace(/^refs\/heads\//, "");
    }
  }

  if (worktree) {
    worktrees.push(worktree);
  }

  return {
    root,
    branch,
    status,
    clean: status.every((line) => line.startsWith("##")),
    worktrees
  };
};

const readFrontmatterValue = (
  frontmatter: string,
  key: string,
  fallbackKey?: string
): string => {
  let match = frontmatter.match(new RegExp(`^${key}:\\s*(.+)$`, "m"));
  if (!match && fallbackKey) {
    match = frontmatter.match(new RegExp(`^${fallbackKey}:\\s*(.+)$`, "m"));
  }

  if (!match) {
    throw new Error(`Agent definition is missing "${key}" metadata.`);
  }

  return match[1].trim().replace(/^(['"])(.*)\1$/, "$2");
};

export const listAgents = (
  repositoryRoot = getRepositoryState().root
): AgentDefinition[] => {
  const antigravityDirectory = path.join(repositoryRoot, ".agents", "agents");
  const githubDirectory = path.join(repositoryRoot, ".github", "agents");

  const useAntigravity =
    fs.existsSync(antigravityDirectory) &&
    fs.readdirSync(antigravityDirectory).some((f) => f.endsWith(".md"));

  const agentsDirectory = useAntigravity ? antigravityDirectory : githubDirectory;
  const relativeDir = useAntigravity
    ? path.posix.join(".agents", "agents")
    : path.posix.join(".github", "agents");

  const files = fs
    .readdirSync(agentsDirectory, { withFileTypes: true })
    .filter((entry) => entry.isFile() && entry.name.endsWith(".md"))
    .map((entry) => entry.name)
    .sort();

  return files.map((file) => {
    const absolutePath = path.join(agentsDirectory, file);
    const content = fs.readFileSync(absolutePath, "utf8");
    const frontmatter = content.match(/^---\r?\n([\s\S]*?)\r?\n---(?:\r?\n|$)/)?.[1];

    if (!frontmatter) {
      throw new Error(`Agent definition has invalid frontmatter: ${file}`);
    }

    return {
      id: file.replace(/(\.agent)?\.md$/, ""),
      name: readFrontmatterValue(frontmatter, "displayName", "name"),
      description: readFrontmatterValue(frontmatter, "description"),
      file: path.posix.join(relativeDir, file)
    };
  });
};

export const findAgentCandidates = (
  request: string,
  repositoryRoot = getRepositoryState().root
): AgentCandidate[] => {
  const normalizedRequest = request.toLocaleLowerCase();

  if (!normalizedRequest.trim()) {
    return [];
  }

  const requestWords = new Set(
    normalizedRequest
      .split(/[^\p{L}\p{N}]+/u)
      .filter((word) => word.length > 2)
  );

  return listAgents(repositoryRoot)
    .map((agent) => {
      const capabilities = agentCapabilities[agent.id] ?? [];
      const matchedCapabilities = capabilities.filter((capability) =>
        normalizedRequest.includes(capability.toLocaleLowerCase())
      );
      const text = `${agent.name} ${agent.description}`.toLocaleLowerCase();
      const descriptionMatches = text
        .split(/[^\p{L}\p{N}]+/u)
        .filter((word) => word.length > 2)
        .filter((word) => requestWords.has(word));
      const score = matchedCapabilities.length * 2 + descriptionMatches.length;

      return {
        ...agent,
        score,
        matchedCapabilities
      };
    })
    .filter((candidate) => candidate.score > 0)
    .sort((left, right) =>
      right.score - left.score || left.name.localeCompare(right.name)
    )
    .slice(0, 5);
};

export const createAgentPlan = (

type Entity = {
  id: string;
};

export const getNextId = (
  prefix: string,
  items: Entity[]
): string => {

  const maxNo =
    items
      .filter(
        item =>
          item.id.startsWith(
            prefix + "-"
          )
      )
      .map(
        item =>
          Number(
            item.id
              .replace(
                prefix + "-",
                ""
              )
          )
      )
      .filter(
        value =>
          !Number.isNaN(
            value
          )
      )
      .reduce(
        (
          current,
          value
        ) =>
          Math.max(
            current,
            value
          ),
        0
      );

  return (
    prefix +
    "-" +
    String(
      maxNo + 1
    ).padStart(
      3,
      "0"
    )
  );
};

export const createIdAllocator =
(
  prefix: string,
  items: Entity[]
) => {

  let current =
    Number(
      getNextId(
        prefix,
        items
      ).replace(
        prefix + "-",
        ""
      )
    );

  return () => {

    const value =
      prefix +
      "-" +
      String(
        current
      ).padStart(
        3,
        "0"
      );

    current++;

    return value;

  };

};

export const MANAGEMENT_DATA_SECTION_TITLE = "管理データ一覧";

export type ManagementDataLink = {
  title: string;
  href: string;
};

const HEADING_PATTERN = /^(#{1,6})\s+(.+?)\s*#*\s*$/;
const FENCE_PATTERN = /^\s*(```|~~~)/;
const LINK_ITEM_PATTERN = /^\s*[-*+]\s+\[([^\]]+)\]\(([^)\s]+)\)\s*$/;

const toHref = (rawTarget: string): string => {
  const withoutAnchor = rawTarget.split("#")[0] ?? "";

  try {
    return decodeURI(withoutAnchor);
  } catch {
    return withoutAnchor;
  }
};

/**
 * README.md から「管理データ一覧」見出し配下の箇条書きリンクを取り出す。
 * 見出しが存在しない場合は undefined、存在してリンクがなければ空配列を返す。
 */
export const parseManagementDataLinks = (
  markdown: string,
): ManagementDataLink[] | undefined => {
  const lines = markdown.replace(/^\uFEFF/, "").split(/\r?\n/);
  const links: ManagementDataLink[] = [];

  let sectionLevel: number | undefined;
  let found = false;
  let inFence = false;

  for (const line of lines) {
    if (FENCE_PATTERN.test(line)) {
      inFence = !inFence;
      continue;
    }

    if (inFence) {
      continue;
    }

    const heading = HEADING_PATTERN.exec(line);

    if (heading) {
      const level = heading[1].length;

      if (sectionLevel !== undefined && level <= sectionLevel) {
        break;
      }

      if (sectionLevel === undefined && heading[2] === MANAGEMENT_DATA_SECTION_TITLE) {
        sectionLevel = level;
        found = true;
      }

      continue;
    }

    if (sectionLevel === undefined) {
      continue;
    }

    const item = LINK_ITEM_PATTERN.exec(line);
    const href = item ? toHref(item[2]) : "";

    if (item && href) {
      links.push({ title: item[1].trim(), href });
    }
  }

  return found ? links : undefined;
};

import { MarkdownList } from "@/models/MarkdownList";
import { BREAK_LINE, LIST_ITEM } from "@/utils/textConst";

/**
 * Markdownのリストをパースする
 * @param content パース対象の文字列
 * @returns パース結果
 */
export const parseLists = (
  content: string
): MarkdownList[] => {

  const lines = content.split(BREAK_LINE);

  const result: MarkdownList[] = [];

  let currentItems: string[] = [];

  let startLine = 0;

  const flush = () => {
    if (currentItems.length === 0) {
      return;
    }

    result.push({
      items: [...currentItems],
      lineNumber: startLine
    });

    currentItems = [];
    startLine = 0;
  };

  lines.forEach((line, index) => {
    const match = line.match(LIST_ITEM);

    if (match) {

      if (currentItems.length === 0) {
        startLine = index + 1;
      }

      currentItems.push(match[2].trim());

      return;
    }

    flush();
  });

  flush();

  return result;
};

import { MarkdownTable } from "@/models/MarkdownTable";
import { BREAK_LINE, TABLE_SEPARATOR } from "@/utils/textConst";

/**
 * マークダウンコンテンツからテーブルを抽出する
 * @param content マークダウンコンテンツ
 * @returns 抽出されたテーブルの配列
 */
export const parseTables = (
  content: string
): MarkdownTable[] => {

  const lines = content.split(BREAK_LINE);
  const tables: MarkdownTable[] = [];

  let index = 0;
  while (
    index < lines.length
  ) {
    const headerLine = lines[index];
    const separatorLine = lines[index + 1];
    const isHeader = headerLine?.includes("|");

    const isSeparator =
      separatorLine &&
      TABLE_SEPARATOR.test(separatorLine);

    if (!isHeader || !isSeparator) {
      index++;
      continue;
    }

    const headers = parseTableRow(headerLine);
    const rows: string[][] = [];

    let rowIndex = index + 2;

    while (
      rowIndex < lines.length &&
      lines[rowIndex].includes("|")
    ) {
      rows.push(parseTableRow(lines[rowIndex]));
      rowIndex++;
    }

    tables.push({
      headers,
      rows,
      lineNumber: index + 1
    });

    index = rowIndex;
  }

  return tables;
};

/**
 * テーブルの行を解析する
 * @param line テーブルの行
 * @returns 解析されたセルの配列
 */
const parseTableRow = (
  line: string
): string[] => {

  return line
    .split("|")
    .map(
      item => item.trim()
    )
    .filter(
      item => item.length > 0
    );
};

const uniqueByTitle = <

  values: T[]

    new Map<
      string,
      T
    >();

  values.forEach(
    value => {

      if (
        map.has(

      map.set(
        value.title,

  );

    ...map.values()
  ];

      heading =>

        TASK_SECTION_KEYWORDS.some(

import type {
  Document
} from "@/models/Document";

import type {
  Task
} from "@/models/Task";

import type {
  Traceability
} from "@/models/Traceability";

export const buildTraceability =
(
  documents: Document[],
  tasks: Task[]
): Traceability[] => {

  const result:
    Traceability[] = [];

  documents.forEach(
    document => {

      tasks.forEach(
        task => {

          const relationType =
            document.type === "REQ"
              ? "REQ-TSK"
              : document.type === "DES"
                ? "DES-TSK"
                : null;

          if (
            !relationType
          ) {
            return;
          }

          result.push({

            sourceId:
              document.id,

            targetId:
              task.id,

            relationType

          });

        }
      );

    }
  );

  return result;

};

  console.log(`Repository: ${repository.root}`);
  console.log(`Branch: ${repository.branch || "(detached HEAD)"}`);
  console.log(`Working tree: ${repository.clean ? "clean" : "has changes"}`);
  console.log("Status:");
  repository.status.forEach((line) => console.log(`  ${line}`));
  console.log("Worktrees:");
  repository.worktrees.forEach(({ path, branch, current }) =>
    console.log(`  ${current ? "* " : "  "}${path} (${branch || "detached HEAD"})`)
  );
};

const printAgents = () => {
  const query = values.join(" ");
  const agents = query
    ? findAgentCandidates(query)
    : listAgents();

  if (jsonOutput) {
    console.log(JSON.stringify(agents, null, 2));
    return;
  }

  agents.forEach((agent) => {
    const score = "score" in agent ? ` [score: ${agent.score}]` : "";
    console.log(`${agent.name}${score}\n  ${agent.description}\n  ${agent.file}`);
  });
  console.log(`Found ${agents.length} agent(s).`);
};

const printPlan = () => {
  const request = values.join(" ");
  const plan = createAgentPlan(request);

  if (jsonOutput) {
    console.log(JSON.stringify(plan, null, 2));
    return;
  }

  console.log(`Request: ${plan.request}`);
  console.log(`Repository: ${plan.repository.root}`);
  console.log(`Branch: ${plan.repository.branch || "(detached HEAD)"}`);
  console.log("Candidate agents:");

  if (plan.candidateAgents.length === 0) {
    console.log("  No matching candidates; select an agent manually.");
  }

  plan.candidateAgents.forEach((agent) =>
    console.log(
      `  ${agent.name} (${agent.score}) - ${agent.matchedCapabilities.join(", ") || agent.description}`
    )
  );
  console.log("Next steps:");
  plan.nextSteps.forEach((step, index) => console.log(`  ${index + 1}. ${step}`));
};

try {
  switch (command) {
    case "status":
      printStatus();
      break;
    case "agents":
      printAgents();
      break;
    case "plan":
      printPlan();
      break;
    case "help":
    case undefined:
      printHelp();
      break;
    default:
      throw new Error(`Unknown command: ${command}`);
  }
} catch (error) {
  console.error(error instanceof Error ? error.message : error);
  process.exitCode = 1;
}

import fs from "fs";
import path from "path";

const IGNORE_DIRS = [

  "node_modules",
  "dist",
  ".git",
  "coverage"

];

const collectFiles =
(
  directory: string
): string[] => {

  const entries =
    fs.readdirSync(
      directory,
      {
        withFileTypes:true
      }
    );

  return entries.flatMap(
    entry => {

      const fullPath =
        path.join(
          directory,
          entry.name
        );

      if (
        entry.isDirectory()
      ) {

        if (
          IGNORE_DIRS.includes(
            entry.name
          )
        ) {
          return [];
        }

        return collectFiles(
          fullPath
        );
      }

      return [fullPath];

    }
  );
};

const exportSource =
(
  targetDir: string,
  outputFile: string
) => {

  const files =
    collectFiles(
      targetDir
    );

  let output = "";

  for (
    const file of files
  ) {

    output +=
      "\n" +
      "=".repeat(80) +
      "\n";

    output +=
      "FILE: " +
      file +
      "\n";

    output +=
      "=".repeat(80) +
      "\n\n";

    output +=
      fs.readFileSync(
        file,
        "utf-8"
      );

    output +=
      "\n\n";
  }

  fs.writeFileSync(
    outputFile,
    output,
    "utf-8"
  );

  console.log(
    "Export Complete:"
    + outputFile
  );
};

const targetDir =
  process.argv[2];

const outputFile =
  process.argv[3] ??
  "source-export.txt";

if (
  !targetDir
) {

  console.error(
    "Usage: export-source <dir>"
  );

  process.exit(1);

}

exportSource(
  targetDir,
  outputFile
);

    return typeof value === "string" && POSITIVE_INTEGER.test(value)
}

export const positiveIntegerFrom = (value: unknown): number | undefined => {
    if (!isPositiveIntegerString(value)) {
        return undefined;
    }

    const parsed = Number(value);
    return Number.isSafeInteger(parsed) ? parsed : undefined;
};

export const convertToPositiveIntegerOrDefault = (value: unknown, defaultValue: number): number | undefined => {
    return positiveIntegerFrom(value) ?? defaultValue;
}

import type { Response } from "express";

const CREATED = 201;
const BAD_REQUEST = 400;
const NOT_FOUND = 404;

export const createResponse = (success: boolean, message?: string, item?: unknown) => {
    const response: Record<string, unknown> = { success };
    if (message) {
        response.message = message;
    }
    if (item) {
        response.item = item;
    }
    return response;
};

export const createSuccessResponse = (res: Response, item?: unknown) => {
    return res.json(createResponse(true, undefined, item));
}

export const createPagedResponse = <T>(
    res: Response,
    paged: {
        items: T[];
        total: number;
        page: number;
        pageSize: number;
    },
) => {
    return res.json({
        success: true,
        ...paged,
    });
};

export const createCreatedResponse = (res: Response, item?: unknown) => {
    return res.status(CREATED).json(createResponse(true, undefined, item));
}

export const createErrorResponse = (res: Response, message: string, errorCode: number) => {
    return res.status(errorCode).json(createResponse(false, message));
};

export const createBadRequestResponse = (res: Response, message: string): Response => {
    return createErrorResponse(res, message, BAD_REQUEST);
};

export const createNotFoundResponse = (res: Response, message: string): Response => {
    return createErrorResponse(res, message, NOT_FOUND);
};
