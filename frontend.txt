import { useEffect, useState } from "react";

import { apiErrorMessage } from "../../services/api";
import { fetchProjectManagementData } from "../../services/projectApi";
import type { ManagementDataIndex } from "../../types/Api";

type Props = {
    projectId: string;
    rootPath: string;
};

export const ManagementDataList = ({ projectId, rootPath }: Props) => {
    const [retry, setRetry] = useState(0);
    const [loaded, setLoaded] = useState<{
        key: string;
        value: ManagementDataIndex;
    }>();
    const [failed, setFailed] = useState<{
        key: string;
        message: string;
    }>();
    // 保存で管理データパスが変わった場合も再取得する
    const requestKey = `${projectId}:${rootPath}:${retry}`;
    const index = loaded?.key === requestKey ? loaded.value : undefined;
    const errorMessage =
        failed?.key === requestKey ? failed.message : "";

    useEffect(() => {
        let active = true;

        fetchProjectManagementData(projectId)
            .then((result) => {
                if (active) setLoaded({ key: requestKey, value: result });
            })
            .catch((cause: unknown) => {
                if (!active) return;
                setFailed({
                    key: requestKey,
                    message: apiErrorMessage(
                        cause,
                        "管理データ一覧を読み込めませんでした",
                    ),
                });
            });

        return () => {
            active = false;
        };
    }, [projectId, requestKey]);

    return (
        <section aria-labelledby="management-data-heading" className="space-y-2">
            <h2 className="text-xl font-bold" id="management-data-heading">
                管理データ一覧
            </h2>

            {!index && !errorMessage && (
                <p role="status">管理データ一覧を読み込み中...</p>
            )}

            {errorMessage && (
                <div className="space-y-2">
                    <p className="text-red-800" role="alert">
                        {errorMessage}
                    </p>
                    <button
                        className="text-blue-700 underline"
                        onClick={() => setRetry((value) => value + 1)}
                        type="button"
                    >
                        再試行
                    </button>
                </div>
            )}

            {index && index.status !== "ok" && (
                <p
                    className="rounded border border-amber-300 bg-amber-50 p-3"
                    role="status"
                >
                    {index.message}
                </p>
            )}

            {index && index.status === "ok" && index.items.length === 0 && (
                <p>管理データ一覧にリンクが登録されていません。</p>
            )}

            {index && index.status === "ok" && index.items.length > 0 && (
                <table className="w-full border-collapse text-left">
                    <caption className="sr-only">
                        管理データパス配下の管理データ一覧
                    </caption>
                    <thead>
                        <tr className="border-b">
                            <th className="py-1 pr-3" scope="col">
                                名称
                            </th>
                            <th className="py-1 pr-3" scope="col">
                                パス
                            </th>
                            <th className="py-1" scope="col">
                                状態
                            </th>
                        </tr>
                    </thead>
                    <tbody>
                        {index.items.map((item) => (
                            <tr className="border-b" key={item.path}>
                                <td className="py-1 pr-3">{item.title}</td>
                                <td className="py-1 pr-3 break-all">
                                    {item.path}
                                </td>
                                <td
                                    className={
                                        item.exists
                                            ? "py-1"
                                            : "py-1 text-red-800"
                                    }
                                >
                                    {item.exists ? "あり" : "ファイルなし"}
                                </td>
                            </tr>
                        ))}
                    </tbody>
                </table>
            )}
        </section>
    );
};

import {
    createContext,
    ReactNode,
    useContext,
    useMemo,
    useState,
} from "react";

import type {
    Project,
} from "../types/Api";

type ProjectContextValue = {
    project?: Project;
    setProject: (
        project?: Project,
    ) => void;
    clearProject: () => void;
};

const ProjectContext =
    createContext<ProjectContextValue>({
        setProject: () => {},
        clearProject: () => {},
    });

export const ProjectProvider = ({
    children,
}: {
    children: ReactNode;
}) => {

    const [
        project,
        setProjectState,
    ] = useState<Project>();

    const value = useMemo(
        () => ({
            project,

            setProject: (
                next?: Project,
            ) =>
                setProjectState(
                    next,
                ),

            clearProject: () =>
                setProjectState(
                    undefined,
                ),
        }),
        [project],
    );

    return (
        <ProjectContext.Provider
            value={value}
        >
            {children}
        </ProjectContext.Provider>
    );
};

export const useProject =
    () =>
        useContext(
            ProjectContext,
        );

    name: "",
    description: "",
    status: "",
    rootPath: "",
};

const listLocationFrom = (value: string | null) =>
    value && /^\/projects(?:\?|$)/.test(value) ? value : "/projects";

export const ProjectDetailPage = () => {
    const { setProject } = useProject();
    const [searchParams, setSearchParams] = useSearchParams();
    const navigate = useNavigate();
    const id = searchParams.get("id");
    const createMode = searchParams.get("mode") === "create";
    const editMode = createMode || searchParams.get("mode") === "edit";
    const returnTo = listLocationFrom(searchParams.get("returnTo"));
    const [loadedProject, setLoadedProject] = useState<{
        id: string;
        value: Project;
    }>();
    const [form, setForm] = useState<ProjectInput>(emptyProject);
    const [loadError, setLoadError] = useState<{
        key: string;
        message: string;
    }>();
    const [saveError, setSaveError] = useState("");
    const [saving, setSaving] = useState(false);
    const [retry, setRetry] = useState(0);
    const project = loadedProject?.id === id ? loadedProject.value : undefined;
    const requestKey = `${createMode ? "create" : (id ?? "missing")}:${retry}`;
    const currentLoadError =
        loadError?.key === requestKey ? loadError.message : "";
    const loading =
        !createMode && !!id && loadedProject?.id !== id && !currentLoadError;

    useEffect(() => {
        let active = true;
        if (createMode || !id) {
            return () => {
                active = false;
            };
        }

        fetchProject(id)
            .then((result) => {
                if (!active) return;
                setLoadedProject({ id, value: result });
                setProject(result);
                setForm({
                    name: result.name,
                    description: result.description,
                    status: result.status,
                    rootPath: result.rootPath,
                });
            })
            .catch((cause: unknown) => {
                if (!active) return;
                setLoadError({
                    key: requestKey,
                    message: apiErrorMessage(
                        cause,
                        "プロジェクトを読み込めませんでした",
                    ),
                });
            });

        return () => {
            active = false;
        };
    }, [createMode, id, requestKey]);

    const originalForm = useMemo(
        () =>
            project
                ? {
                      name: project.name,
                      description: project.description,
                      status: project.status,
                      rootPath: project.rootPath,
                  }
                : emptyProject,
        [project],
    );
    const dirty = JSON.stringify(form) !== JSON.stringify(originalForm);

        if (createMode) {
            navigate(returnTo);
            return;
        }

        const next = new URLSearchParams(searchParams);
        next.delete("mode");
        setSearchParams(next);
    };

    const save = async (event: FormEvent<HTMLFormElement>) => {

        if (!form.name.trim()) {

                    type="button"

                    プロジェクト一覧へ戻る

    const formKey = createMode ? "create" : id || "missing";
    const [loadedTask, setLoadedTask] = useState<{
        id: string;
        value: Task;
    }>();
    const [formState, setFormState] = useState<{
        key: string;
        value: TaskInput;
    }>({ key: "", value: emptyTask });
    const form =
        formState.key === formKey ? formState.value : emptyTask;

        retry: number;

        error: string;

    const currentProjectState =
        projectState.retry === projectRetry ? projectState : undefined;

    const projectError = currentProjectState?.error || "";
    const [loadError, setLoadError] = useState<{
        key: string;
        message: string;
    }>();
    const [saveError, setSaveError] = useState("");
    const [saving, setSaving] = useState(false);
    const task = loadedTask?.id === id ? loadedTask.value : undefined;
    const requestKey = `${createMode ? "create" : id || "missing"}:${retry}`;
    const currentLoadError =
        loadError?.key === requestKey ? loadError.message : "";
    const loading =
        !createMode && !!id && loadedTask?.id !== id && !currentLoadError;

    useEffect(() => {
        let active = true;
        if (createMode || !id) {
            return () => {
                active = false;
            };
        }

        fetchTask(id)

                if (!active) return;
                setLoadedTask({ id, value: result });
                setFormState({ key: id, value: formFromTask(result) });
            })
            .catch((cause: unknown) => {
                if (!active) return;
                setLoadError({
                    key: requestKey,
                    message: apiErrorMessage(
                        cause,
                        "タスクを読み込めませんでした",
                    ),
                });
            });

export const TaskListPage = () => {

    const searchKey = searchParams.toString();
    const filters = useMemo<TaskListFilters>(() => {

        return {
            q: params.get("q") || "",
            projectId,
            status: params.get("status") || undefined,
            priority: params.get("priority") || undefined,
            page: parsePage(params.get("page")),

    }, [searchKey]);
    const [draftState, setDraftState] = useState<{
        key: string;
        value: TaskListFilters;
    }>({ key: "", value: filters });
    const draft =
        draftState.key === searchKey ? draftState.value : filters;

        setDraftState({
            key: searchKey,
            value: { ...draft, ...updates },
        });
    };
    const [retry, setRetry] = useState(0);
    const [projectRetry, setProjectRetry] = useState(0);

        error: string;

    const listError = currentTaskState?.error || "";
    const loading = !currentTaskState;

        error: string;

import axios from "axios";

export const api = axios.create({
  baseURL: "http://localhost:3000/api"
});

export const apiErrorMessage = (
  error: unknown,
  fallback: string,
): string => {
  if (axios.isAxiosError<{ message?: unknown }>(error)) {
    const message = error.response?.data?.message;
    if (typeof message === "string" && message.trim()) {
      return message;
    }
    return error.message || fallback;
  }

  return error instanceof Error && error.message
    ? error.message
    : fallback;
};
